\section{Proofs for Task-Relevant Information Preservation}
\label{app:representationproofs}

% \subsection{专家动作重构界的证明}
\subsection{Proof of the Expert-Action Reconstruction Bound}
\label{app:expertreconstruction}
% 固定两种分布与共同读出，用三角不等式证明正文重构界。
We prove~\eqref{eq:expertreconstruction} under the common encoder and readouts of Section~\ref{sec:representationanalysis}, with $\mu=\bar\mu_\pi$.
% 第一步：共同的专家动作近似是Z的可测函数。
Define the common represented expert rule $\widehat a_{\mathrm E}(Z)=g_{\mathrm E}(\widehat G,\chi)$.
% 三角不等式和专家选择假设分别约束两分布上的动作重构误差。
The triangle inequality and Assumption~\ref{ass:expertselection} give, for $\rho\in\{\mu,\nu\}$,
\begin{equation}
 \begin{aligned}
 \E_\rho\norm{\pi_{\mathrm E}-\widehat a_{\mathrm E}}
 &\le \E_\rho\norm{\pi_{\mathrm E}-g_{\mathrm E}(G,\chi)}\\
 &\quad+\E_\rho\norm{g_{\mathrm E}(G,\chi)-g_{\mathrm E}(\widehat G,\chi)}\\
 &\le\epsilon_{\mathrm{sel},\rho}+K_{\mathrm E}\delta_{G,\rho}
 =b_\rho.
 \end{aligned}
 \label{eq:appexpertreconstruction}
\end{equation}


% \subsection{条件误差迁移界的证明}
\subsection{Proof of the Conditional-Error Transfer Bound}
\label{app:relationtransfer}
% 先对比真实动作误差与表示中的共同函数，再用覆盖条件合并两分布的重构误差。
We prove Proposition~\ref{prop:relationtransfer} by comparing actual imitation error with a common function of $Z$, then transferring the demonstration-side reconstruction error through representation coverage.
% 沿用正文的策略和专家读出。
Use $\pi(h)=p(Z)$ and $\widehat a_{\mathrm E}(Z)=g_{\mathrm E}(\widehat G,\chi)$ from Section~\ref{sec:representationanalysis}.
% 第二步：策略和专家近似均只依赖Z，因此其平方距离在两分布中是同一个函数。
Since $\pi(h)=p(\phi(t,h))$, the represented squared error
$\ell_Z(z)=\norm{p(z)-\widehat a_{\mathrm E}(z)}^2$ is the same function of $z$ under both distributions.
% 两个动作距离之和不超过两倍动作集合直径，反三角不等式约束距离之差。
All three actions belong to $\mathcal A$, so the reverse triangle inequality yields
\begin{equation}
 \begin{aligned}
 |e(h)^2-\ell_Z(Z)|
 &\le\bigl(\norm{\pi-\pi_{\mathrm E}}
       +\norm{\pi-\widehat a_{\mathrm E}}\bigr)
       \norm{\pi_{\mathrm E}-\widehat a_{\mathrm E}}\\
 &\le 2B_{\mathcal A}\norm{\pi_{\mathrm E}-\widehat a_{\mathrm E}}.
 \end{aligned}
 \label{eq:appsquarederrorperturbation}
\end{equation}
% 分别在两分布下条件化，公共函数ell_Z留在条件期望之外。
Conditioning separately under each distribution gives
\begin{equation}
 |m_\rho(z)-\ell_Z(z)|
 \le 2B_{\mathcal A}
 \E_\rho[\norm{\pi_{\mathrm E}-\widehat a_{\mathrm E}}\mid Z=z].
 \label{eq:appconditionalexpertapprox}
\end{equation}

% 第三步：插入相同的ell_Z，用表示覆盖迁移示范侧条件误差。
Insert the common function $\ell_Z$ between $m_\mu$ and $m_\nu$ and apply representation coverage to the nonnegative demonstration-side conditional error:
\begin{equation}
 \begin{aligned}
 D_\phi
 &\le\E_{\mu_Z}|m_\mu(Z)-\ell_Z(Z)|\\
 &\quad+\E_{\mu_Z}|m_\nu(Z)-\ell_Z(Z)|\\
 &\le\E_{\mu_Z}|m_\mu(Z)-\ell_Z(Z)|\\
 &\quad+C_\phi\E_{\nu_Z}|m_\nu(Z)-\ell_Z(Z)|\\
 &\le 2B_{\mathcal A}(b_\mu+C_\phi b_\nu).
 \end{aligned}
 \label{eq:apprelationdiscrepancy}
\end{equation}
% 有限表示覆盖使示范侧条件不等式也在部署表示分布下几乎处处成立。
Representation coverage ensures that the demonstration-side conditional inequality also holds $\mu_Z$-almost everywhere.
% 代入B3的任务界即可，无需重复完整结论。
This proves~\eqref{eq:relationdiscrepancybound}; substitution into~\eqref{eq:representationcoveragebound} establishes~\eqref{eq:relationtransferfailure}.
\section{Proofs for Information-Conditioned Residual Corrections}
\label{app:feedbackproofs}
% 三个证明均固定基础策略、专家目标和示范总体分布nu。
All proofs fix the base policy, expert targets, and distribution $\nu$ from Section~\ref{sec:feedbackanalysis}.

% \subsection{最优残差预测的证明}
\subsection{Proof of the Optimal Residual Predictor}
\label{app:optimalresidual}
% 条件期望的正交性使任意可用信息函数与不可预测误差不相关。
For any square-integrable $\mathcal F$-measurable vector $v$, the tower property gives
\begin{equation}
 \begin{aligned}
 \E_\nu\inner{E-r_{\mathcal F}^*}{v}
 &=\E_\nu\inner{\E_\nu[E-r_{\mathcal F}^*\mid\mathcal F]}{v}\\
 &=0.
 \end{aligned}
 \label{eq:residualorthogonality}
\end{equation}
% 将误差拆成条件均值残差与预测偏差，交叉项为零。
For any square-integrable $\mathcal F$-measurable predictor $r$, write
$E-r=(E-r_{\mathcal F}^*)+(r_{\mathcal F}^*-r)$.
Expanding the square and applying~\eqref{eq:residualorthogonality} yields
\begin{equation}
 \E_\nu\norm{E-r}^2
 =R_\nu^*(\mathcal F)+\E_\nu\norm{r-r_{\mathcal F}^*}^2.
 \label{eq:residualprojection}
\end{equation}
% 非负的后一项说明条件均值实现无约束预测误差的最小值。
The second term is nonnegative and vanishes at $r=r_{\mathcal F}^*$, proving~\eqref{eq:optimalresidual}.

% \subsection{信息收益恒等式的证明}
\subsection{Proof of the Information-Gain Identity}
\label{app:residualinformationgain}
% 嵌套信息允许把较少信息下的最优预测当作较多信息下的候选预测。
For $\mathcal F_0\subseteq\mathcal F_1$, the tower property gives
$\E_\nu[r_{\mathcal F_1}^*\mid\mathcal F_0]=r_{\mathcal F_0}^*$.
Since $r_{\mathcal F_0}^*$ is also $\mathcal F_1$-measurable, applying~\eqref{eq:residualprojection} with information $\mathcal F_1$ and predictor $r_{\mathcal F_0}^*$ gives
\begin{equation}
 R_\nu^*(\mathcal F_0)
 =R_\nu^*(\mathcal F_1)
  +\E_\nu\norm{r_{\mathcal F_1}^*-r_{\mathcal F_0}^*}^2.
 \label{eq:appresidualinformationgain}
\end{equation}
% 移项即得正文结论；只有条件均值在正概率情形改变时才严格改善。
Rearranging proves~\eqref{eq:residualinformationgain}.
The gain is strictly positive exactly when the two conditional means differ on a set of positive $\nu$-probability.

% \subsection{实际纠偏条件的证明}
\subsection{Proof of the Realized-Correction Criterion}
\label{app:realizedresidual}
% 零残差对应基础策略，代入投影恒等式得到其误差分解。
Setting $r=0$ in~\eqref{eq:residualprojection} gives
\begin{equation}
 R_\nu(\pi_{\mathrm b})
 =R_\nu^*(\mathcal F)+\E_\nu\norm{r_{\mathcal F}^*}^2.
 \label{eq:basepredictableerror}
\end{equation}
% 再代入学得残差，两式相减即可得到正文实际收益公式。
Setting $r=r_{\mathcal F}$ in the same identity gives
$R_\nu(\pi_{\mathcal F})=R_\nu^*(\mathcal F)+\epsilon_{\mathrm{res}}(\mathcal F)$.
Combining these expressions proves~\eqref{eq:realizedresidualrisk}.
% 两个信息条件下的实际差异由潜在信息收益和学习误差差异组成。
For two learned corrections under nested information, subtraction further gives
\begin{equation}
 \begin{aligned}
 R_\nu(\pi_{\mathcal F_0})-R_\nu(\pi_{\mathcal F_1})
 &=\E_\nu\norm{r_{\mathcal F_1}^*-r_{\mathcal F_0}^*}^2\\
 &\quad+\epsilon_{\mathrm{res}}(\mathcal F_0)
       -\epsilon_{\mathrm{res}}(\mathcal F_1).
 \end{aligned}
 \label{eq:learnedinformationgain}
\end{equation}
% 信息收益必须超过新增实现误差才会改善，而非由增加历史或网络结构直接保证。
Thus, the predictive gain must exceed any additional realization error for richer information to improve the learned policy.

% 展开平方提供无需知道理想预测器的同分布误差诊断。
Equivalently, direct expansion gives an error diagnostic without estimating the ideal predictor:
\begin{equation}
 R_\nu(\pi_{\mathrm b})-R_\nu(\pi_{\mathcal F})
 =2\E_\nu\inner E{r_{\mathcal F}}
  -\E_\nu\norm{r_{\mathcal F}}^2.
 \label{eq:residualgain}
\end{equation}
% 此处残差是实际施加值，含限幅和未采用时的零修正；不把候选误差替代执行误差。
Here $r_{\mathcal F}$ includes output processing and is zero when no correction is applied; it must remain measurable with respect to the available information.
% 各完整策略使用其自身的部署分布与表示迁移量，不能将它们因纠偏而发生的变化忽略。
For admissible actions, apply~\eqref{eq:representationcoveragebound} to each complete policy using its own deployment distribution and representation.
A reduction in $R_\nu$ tightens that bound when the other terms are held fixed, or more generally when $C_\phi R_\nu+D_\phi$ decreases with a common expert baseline, horizon, and positive sensitivity bound.
% 总体分布上的恒等式不直接给出有限训练损失或闭环成功率保证。
The identities concern population error under $\nu$; replacing it with training loss requires a separate generalization argument.
